// ==================== User Course Progress ====================
// Collection: user_course_progress
// Document ID format: {userId}_{courseId}

match /user_course_progress/{progressId} {
  // Extract userId and courseId from document ID
  function getUserIdFromProgress() {
    return progressId.split('_')[0];
  }
  
  function getCourseIdFromProgress() {
    return progressId.split('_')[1];
  }
  
  // Check if user is enrolled in the course
  function isEnrolled(userId, courseId) {
    let course = get(/databases/$(database)/documents/courses/$(courseId));
    return userId in course.data.trainees || userId == course.data.trainerId;
  }
  
  // Check if user is trainer of the course
  function isTrainer(userId, courseId) {
    let course = get(/databases/$(database)/documents/courses/$(courseId));
    return userId == course.data.trainerId;
  }
  
  // Read: User can read their own progress, trainer can read all
  allow read: if request.auth != null && (
    request.auth.uid == getUserIdFromProgress() ||
    isTrainer(request.auth.uid, getCourseIdFromProgress())
  );
  
  // Create: Only system (via service) can create
  // Progress is created automatically when user earns first points
  allow create: if request.auth != null &&
    request.auth.uid == getUserIdFromProgress() &&
    isEnrolled(request.auth.uid, getCourseIdFromProgress()) &&
    request.resource.data.userId == request.auth.uid &&
    request.resource.data.courseId == getCourseIdFromProgress();
  
  // Update: Only system (via service) can update
  // Ensures data integrity - points can only be added through service
  allow update: if request.auth != null &&
    request.auth.uid == getUserIdFromProgress() &&
    request.resource.data.userId == resource.data.userId &&
    request.resource.data.courseId == resource.data.courseId;
  
  // Delete: Only trainer can delete (for reset functionality)
  allow delete: if request.auth != null &&
    isTrainer(request.auth.uid, getCourseIdFromProgress());
}

// ==================== Points Transactions ====================
// Collection: points_transactions
// Tracks all point awards for audit trail

match /points_transactions/{transactionId} {
  // Read: User can read their own transactions, trainer can read all for their course
  allow read: if request.auth != null && (
    request.auth.uid == resource.data.userId ||
    isTrainerOfCourse(request.auth.uid, resource.data.courseId)
  );
  
  // Create: Only system (via service) can create transactions
  // Transactions are created when points are awarded
  allow create: if request.auth != null &&
    request.auth.uid == request.resource.data.userId &&
    request.resource.data.points is int &&
    request.resource.data.points > 0 &&
    request.resource.data.timestamp is timestamp;
  
  // Update/Delete: Never allowed - transactions are immutable audit log
  allow update, delete: if false;
}

// ==================== Gamification Settings ====================
// Embedded in courses collection as 'gamificationSettings' field

match /courses/{courseId} {
  // These rules extend your existing course rules
  
  // Read gamificationSettings: Anyone enrolled in course
  allow read: if request.auth != null && (
    request.auth.uid in resource.data.trainees ||
    request.auth.uid == resource.data.trainerId
  );
  
  // Update gamificationSettings: Only trainer
  // Must preserve other course fields
  allow update: if request.auth != null &&
    request.auth.uid == resource.data.trainerId &&
    // Ensure gamificationSettings is the only thing changing
    (
      !request.resource.data.diff(resource.data).affectedKeys().hasAny(
        ['trainees', 'trainerId', 'title', 'description', 'createdAt']
      )
    );
}
